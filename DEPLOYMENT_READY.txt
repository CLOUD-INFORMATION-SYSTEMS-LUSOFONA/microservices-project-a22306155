PASSO 4: CONFIGURAR ANSIBLE INVENTORY (2 min)
───────────────────────────────────────────────
Edit: ansible/inventory.ini

Adicionar EC2 IP:
  [web_servers]
  ec2_prod ansible_host=54.123.45.67 ansible_user=ec2-user ansible_ssh_private_key_file=~/.ssh/<key>.pem

PASSO 5: RUN ANSIBLE PLAYBOOKS (20-30 min)
────────────────────────────────────────────

# Playbook 1: Setup EC2 (Docker, users, directories)
ansible-playbook -i ansible/inventory.ini ansible/playbooks/configure-ec2.yml -vvv

# Playbook 2: Deploy serviços (docker-compose)
ansible-playbook -i ansible/inventory.ini ansible/playbooks/deploy-app.yml \
  --extra-vars "rds_endpoint=<RDS_ENDPOINT> db_password=<PASSWORD> sqs_queue_url=<SQS_URL>" \
  -vvv

PASSO 6: VERIFICAR SERVIÇOS (5-10 min)
───────────────────────────────────────

# SSH para EC2 (verificar que tudo está running)
ssh -i ~/.ssh/<key>.pem ec2-user@<EC2_IP>

# Verificar containers
docker compose -f /opt/app/docker-compose.yml ps

# Ver logs
docker compose -f /opt/app/docker-compose.yml logs

# Test health endpoints (local machine)
curl http://<EC2_IP>:8080/actuator/health
curl http://<EC2_IP>:8081/actuator/health  (user-service)
curl http://<EC2_IP>:8082/actuator/health  (product-service)
